import { execSync }                             from 'child_process'
import { existsSync, statSync, readdirSync }    from 'fs'
import { join }                                  from 'path'

export async function buildProject(): Promise<void> {
  const cwd = process.cwd()

  console.log('\n\x1b[36m[AxilJS]\x1b[0m Building for production...\n')

  if (!existsSync(join(cwd, 'tsconfig.json'))) {
    console.error('\x1b[31mError:\x1b[0m No tsconfig.json found.')
    process.exit(1)
  }

  const start = Date.now()

  console.log('  \x1b[90mType checking...\x1b[0m')
  try {
    execSync('npx tsc --noEmit', { cwd, stdio: 'pipe' })
    console.log('  \x1b[32m✓\x1b[0m Type check passed')
  } catch (error: any) {
    console.error('  \x1b[31m✗\x1b[0m Type check failed:\n')
    console.error(error.stdout?.toString() || error.message)
    process.exit(1)
  }

  console.log('  \x1b[90mCompiling TypeScript...\x1b[0m')
  try {
    execSync('npx tsc', { cwd, stdio: 'pipe' })
    console.log('  \x1b[32m✓\x1b[0m Compilation complete')
  } catch (error: any) {
    console.error('  \x1b[31m✗\x1b[0m Compilation failed:\n')
    console.error(error.stdout?.toString() || error.message)
    process.exit(1)
  }

  const distDir = join(cwd, 'dist')
  if (existsSync(distDir)) {
    const size     = getDirSize(distDir)
    const duration = Date.now() - start

    console.log('')
    console.log(`\x1b[32m✓\x1b[0m Build complete in \x1b[1m${duration}ms\x1b[0m`)
    console.log(`  Output: \x1b[36mdist/\x1b[0m (${formatBytes(size)})`)
    console.log('')
    console.log('  Start production server:')
    console.log('  \x1b[90mnode dist/main.js\x1b[0m')
    console.log('')
  }
}

function getDirSize(dirPath: string): number {
  let total = 0
  const items = readdirSync(dirPath, { withFileTypes: true })
  for (const item of items) {
    const itemPath = join(dirPath, item.name)
    if (item.isDirectory()) total += getDirSize(itemPath)
    else total += statSync(itemPath).size
  }
  return total
}

function formatBytes(bytes: number): string {
  if (bytes < 1024)        return `${bytes} B`
  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`
}

function prompt(question: string, fallback: string): Promise<string> {
  const rl = createInterface({ input: process.stdin, output: process.stdout })

  return new Promise((res) => {
    const q = `  ${s('?', ANSI.cyan + ANSI.bold)}  ${question} ${s(`(${fallback})`, ANSI.gray)} `
    rl.question(q, (raw) => {
      rl.close()
      res(raw.trim() || fallback)
    })
  })
}

function confirmPrompt(question: string, fallback = false): Promise<boolean> {
  const hint = fallback
    ? s('Y', ANSI.green + ANSI.bold) + s('/n', ANSI.gray)
    : s('y', ANSI.gray) + s('/N', ANSI.white + ANSI.bold)

  const rl = createInterface({ input: process.stdin, output: process.stdout })

  return new Promise((res) => {
    const q = `  ${s('?', ANSI.cyan + ANSI.bold)}  ${question} ${hint} `
    rl.question(q, (raw) => {
      rl.close()
      const a = raw.trim().toLowerCase()
      if (['y', 'yes'].includes(a)) return res(true)
      if (['n', 'no'].includes(a))  return res(false)
      res(fallback)
    })
  })
}

function validateProjectName(name: string): void {
  const n = name.trim()
  if (!n) throw new Error('Project name is required.')
  if (n === '.' || n === '..' || n.includes('/') || n.includes('\\'))
    throw new Error('Project name must be a single directory name, not a path.')
  if (!/^[a-zA-Z0-9._-]+$/.test(n))
    throw new Error('Only letters, numbers, dots, hyphens, and underscores are allowed.')
  if (n.startsWith('.') || n.startsWith('-'))
    throw new Error('Cannot start with "." or "-".')
}

function writeFile(dir: string, rel: string, content: string): void {
  const fp = join(dir, rel)
  mkdirSync(resolve(fp, '..'), { recursive: true })
  writeFileSync(fp, content, 'utf8')
}

function run(cmd: string, args: string[], cwd: string): void {
  const r = spawnSync(cmd, args, {
    cwd,
    stdio: 'inherit',
    shell: process.platform === 'win32'
  })
  if (r.error) throw r.error
  if (r.status !== 0)
    throw new Error(`${cmd} ${args.join(' ')} exited with code ${r.status ?? 1}.`)
}

function getAxilJSVersion(): string {
  const p = path.resolve(__dirname, '../../package.json')
  if (!fs.existsSync(p))
    throw new Error(`Cannot find CLI package.json at: ${p}`)
  const pkg = JSON.parse(fs.readFileSync(p, 'utf-8'))
  if (!pkg.version)
    throw new Error(`CLI package.json has no version field: ${p}`)
  return pkg.version as string
}

function generatePackageJson(name: string, options: ProjectOptions): string {
  const deps: Record<string, string> = {
    '@axiljs/core':   `^${axilVersion}`,
    '@axiljs/config': `^${axilVersion}`
  }

  if (options.database !== 'none') {
    deps['@axiljs/orm'] = `^${axilVersion}`
    if (options.database === 'postgres') deps['pg']             = '^8.23.0'
    if (options.database === 'mysql')    deps['mysql2']         = '^3.23.3'
    if (options.database === 'sqlite')   deps['better-sqlite3'] = '^13.0.3'
  }

  if (options.includeAuth)      deps['@axiljs/auth']      = `^${axilVersion}`
  if (options.includeAI)        deps['@axiljs/ai']        = `^${axilVersion}`
  if (options.includeWebsocket) deps['@axiljs/websocket'] = `^${axilVersion}`
  if (options.includeQueue)     deps['@axiljs/queue']     = `^${axilVersion}`

  return JSON.stringify({
    name,
    version: '1.0.0',
    private: true,
    type: 'module',
    description: `${name} - AxilJS application`,
    scripts: {
      dev:   'axil run',
      build: 'tsc',
      start: 'node dist/main.js',
      test:  'axil test'
    },
    dependencies: deps,
    devDependencies: {
      '@axiljs/cli': `^${axilVersion}`,

function generateUsersService(options: ProjectOptions): string {
  if (options.database === 'none') {
    return `/**
 * Users Service.
 * In-memory implementation (no database configured).
 */

import { NotFoundError } from '@axiljs/core'

export interface User {
  id: string; name: string; email: string; createdAt: string
}

export class UsersService {
  private users: User[] = [
    { id: '1', name: 'Example User', email: 'user@example.com', createdAt: new Date().toISOString() }
  ]

  async findAll(): Promise<User[]> { return this.users }

  async findById(id: string): Promise<User> {
    const u = this.users.find((x) => x.id === id)
    if (!u) throw new NotFoundError(\`User \${id} not found.\`)
    return u
  }

  async create(data: Partial<User>): Promise<User> {
    const u: User = {
      id: String(this.users.length + 1),
      name: data.name || '', email: data.email || '',
      createdAt: new Date().toISOString()
    }
    this.users.push(u)
    return u
  }

  async update(id: string, data: Partial<User>): Promise<User> {
    const i = this.users.findIndex((x) => x.id === id)
    if (i === -1) throw new NotFoundError(\`User \${id} not found.\`)
    this.users[i] = { ...this.users[i], ...data }
    return this.users[i]
  }

  async delete(id: string): Promise<void> {
    const i = this.users.findIndex((x) => x.id === id)
    if (i === -1) throw new NotFoundError(\`User \${id} not found.\`)
    this.users.splice(i, 1)
  }
}
`
  }

  const p = options.database === 'postgres' ? '$1' : '?'
  return `/**
 * Users Service.
 * Database-backed implementation.
 */

import { NotFoundError } from '@axiljs/core'
import { getDB } from '../../../db/connection.js'

export interface User {
  id: string; name: string; email: string; createdAt: string
}

function toUser(row: Record<string, unknown>): User {
  return {
    id: String(row.id), name: String(row.name ?? ''),
    email: String(row.email ?? ''),
    createdAt: String(row.createdAt ?? row.created_at ?? '')
  }
}

export class UsersService {
  private get driver() { return getDB().getDriver() }

  async findAll(): Promise<User[]> {
    const r = await this.driver.query('SELECT * FROM users ORDER BY id DESC')
    return r.rows.map(toUser)
  }

  async findById(id: string): Promise<User> {
    const r = await this.driver.query('SELECT * FROM users WHERE id = ${p}', [id])
    if (r.rows.length === 0) throw new NotFoundError(\`User \${id} not found.\`)
    return toUser(r.rows[0])
  }

  async create(data: Partial<User>): Promise<User> {
    const r = await this.driver.query(
      'INSERT INTO users (name, email) VALUES (${p}, ${p}) RETURNING *',
      [data.name || '', data.email || '']
    )
    return toUser(r.rows[0])
  }

  async update(id: string, data: Partial<User>): Promise<User> {
    const r = await this.driver.query(
      'UPDATE users SET name = COALESCE(${p}, name), email = COALESCE(${p}, email) WHERE id = ${p} RETURNING *',
      [data.name, data.email, id]
    )
    if (r.rows.length === 0) throw new NotFoundError(\`User \${id} not found.\`)
    return toUser(r.rows[0])
  }

  async delete(id: string): Promise<void> {
    const r = await this.driver.query('DELETE FROM users WHERE id = ${p} RETURNING id', [id])
    if (r.rows.length === 0) throw new NotFoundError(\`User \${id} not found.\`)
  }
}
`.replaceAll('${p}', p)
}

import { readFileSync, writeFileSync, mkdirSync, existsSync } from 'fs'
import { join } from 'path'

export async function deploy(): Promise<void> {
  console.log('\n\x1b[36m[AxilJS Deploy]\x1b[0m Generating deployment artifacts...\n')

  const cwd = process.cwd()
  const pkgPath = join(cwd, 'package.json')

  if (!existsSync(pkgPath)) {
    console.error('\x1b[31mError:\x1b[0m No package.json found.')
    process.exit(1)
  }

  let pkg: any
  try { pkg = JSON.parse(readFileSync(pkgPath, 'utf-8')) }
  catch {
    console.error('\x1b[31mError:\x1b[0m Could not parse package.json')
    process.exit(1)
  }

  const appName  = pkg.name    || 'axil-app'
  const version  = pkg.version || '1.0.0'
  const port     = 3000
  const replicas = 3

  const dir = join(cwd, 'deploy')
  mkdirSync(dir, { recursive: true })

  // Dockerfile
  writeFileSync(join(dir, 'Dockerfile'), `# AxilJS Production Dockerfile
FROM node:20-alpine AS builder
WORKDIR /app
COPY package*.json ./
RUN npm ci
COPY . .
RUN npm run build

FROM node:20-alpine AS production
WORKDIR /app
COPY package*.json ./
RUN npm ci --production
COPY --from=builder /app/dist ./dist
EXPOSE ${port}
ENV NODE_ENV=production
CMD ["node", "dist/main.js"]
`)

  // Kubernetes manifest
  writeFileSync(join(dir, 'deployment.yaml'), `apiVersion: apps/v1
kind: Deployment
metadata:
  name: ${appName}
spec:
  replicas: ${replicas}

    matchLabels:
      app: ${appName}
  template:
    metadata:
      labels:
        app: ${appName}
    spec:
      containers:
      - name: ${appName}
        image: ${appName}:${version}
        ports:
        - containerPort: ${port}
        env:
        - name: NODE_ENV
          value: "production"
        - name: PORT
          value: "${port}"
        livenessProbe:
          httpGet:
            path: /health/live
            port: ${port}
          initialDelaySeconds: 10
          periodSeconds: 30
        readinessProbe:
          httpGet:
            path: /health/ready
            port: ${port}
          initialDelaySeconds: 5
          periodSeconds: 10
---
apiVersion: v1
kind: Service
metadata:
  name: ${appName}
spec:
  selector:
    app: ${appName}
  ports:
  - port: 80
    targetPort: ${port}
`)

  // Deploy script
  writeFileSync(join(dir, 'deploy.sh'), `#!/bin/bash
set -e
APP="${appName}"
VERSION="${version}"
echo "Building $APP:$VERSION..."
docker build -t $APP:$VERSION .
echo "Applying Kubernetes manifests..."

  console.log('Next steps:\n')
  console.log('  npm run build')
  console.log(`  docker build -t ${appName} .`)

/**
 * axil doctor
 * Diagnoses project for common issues.
 */

import { existsSync, readFileSync } from 'fs'
import { join }                     from 'path'
import { execSync }                 from 'child_process'

interface Check {
  name: string
  pass: boolean
  message: string
}

export async function runDoctor(): Promise<void> {
  console.log('\n\x1b[36m[AxilJS]\x1b[0m Running diagnostics...\n')

  const checks: Check[] = []
  const cwd = process.cwd()

  // Node.js version
  const nodeVersion = process.version
  const nodeMajor   = parseInt(nodeVersion.slice(1))
  checks.push({
    name:    'Node.js version',
    pass:    nodeMajor >= 18,
    message: `${nodeVersion} ${nodeMajor >= 18 ? '(OK)' : '(need >= 18)'}`
  })

  // TypeScript
  let tsVersion = 'not found'
  let hasTs     = false
  try {
    tsVersion = execSync('npx tsc --version', { stdio: 'pipe' }).toString().trim()
    hasTs = true
  } catch {}
  checks.push({ name: 'TypeScript', pass: hasTs, message: tsVersion })

  // Project files
  const requiredFiles = [
    { file: 'package.json',  name: 'package.json' },
    { file: 'tsconfig.json', name: 'tsconfig.json' },
    { file: 'src/main.ts',   name: 'src/main.ts (entry point)' }
  ]

  for (const { file, name } of requiredFiles) {
    const exists = existsSync(join(cwd, file))
    checks.push({ name, pass: exists, message: exists ? 'Found' : 'Missing' })
  }

  // Check for @axiljs/core
  const pkgPath = join(cwd, 'package.json')
  if (existsSync(pkgPath)) {
    const pkg     = JSON.parse(readFileSync(pkgPath, 'utf-8'))
    const hasCore = pkg.dependencies?.['@axiljs/core'] !== undefined ||
                    pkg.devDependencies?.['@axiljs/core'] !== undefined

    checks.push({
      name:    '@axiljs/core dependency',
      pass:    hasCore,
      message: hasCore
        ? pkg.dependencies?.['@axiljs/core'] || pkg.devDependencies?.['@axiljs/core']
        : 'Not found in dependencies'
    })

    // Lock file
    const hasLock = existsSync(join(cwd, 'axil.lock'))
    checks.push({
      name:    'axil.lock (deterministic installs)',
      pass:    hasLock,
      message: hasLock ? 'Found' : 'Missing — run axil install'
    })
  }

  // .env protection
  const hasGitignore  = existsSync(join(cwd, '.gitignore'))
  let   envProtected  = false

  if (hasGitignore) {
    const gitignore = readFileSync(join(cwd, '.gitignore'), 'utf-8')
    envProtected    = gitignore.includes('.env')
  }

  checks.push({
    name:    '.env protected in .gitignore',
    pass:    envProtected,
    message: envProtected ? 'Protected' : '.env not in .gitignore'
  })

  // Print results
  const passed = checks.filter(c => c.pass).length
  const failed = checks.filter(c => !c.pass).length

  for (const check of checks) {
    const icon = check.pass ? '\x1b[32m✓\x1b[0m' : '\x1b[31m✗\x1b[0m'
    console.log(`  ${icon} ${check.name.padEnd(35)} ${check.message}`)
  }

  console.log('')
  console.log(
    `  ${passed}/${checks.length} checks passed` +
    (failed > 0 ? ` \x1b[31m(${failed} issues found)\x1b[0m` : ' \x1b[32m✓\x1b[0m')
  )
  console.log('')
}

  const cwd = process.cwd()

  if (packages.length > 0) {
    await addToPackageJson(packages, isDev, cwd)
  }

  // Delegate to npm for actual installation
  const { execSync } = require('child_process')
  try {
    execSync('npm install', { cwd, stdio: 'inherit' })
  } catch {
    console.error('\x1b[31m[AxilJS]\x1b[0m Installation failed.')
    process.exit(1)
  }
}

async function addToPackageJson(
  packages: string[],
  isDev: boolean,
  cwd: string
): Promise<void> {
  const pkgPath = join(cwd, 'package.json')

  if (!existsSync(pkgPath)) {
    console.error('\x1b[31mError:\x1b[0m No package.json found.')
    process.exit(1)
  }

  const pkg     = JSON.parse(readFileSync(pkgPath, 'utf-8'))
  const section = isDev ? 'devDependencies' : 'dependencies'

  if (!pkg[section]) pkg[section] = {}

  for (const packageName of packages) {
    const atIndex = packageName.lastIndexOf('@')
    let name    = packageName
    let version = 'latest'

    if (atIndex > 0) {
      name    = packageName.slice(0, atIndex)
      version = packageName.slice(atIndex + 1)
    }

    pkg[section][name] = `^${version}`
    console.log(`  \x1b[32m+\x1b[0m Added ${name}@${version} to ${section}`)
  }

  writeFileSync(pkgPath, JSON.stringify(pkg, null, 2), 'utf-8')
}

/**
 * axil run
 * Development server with hot reload.
 * Watches src/ for TypeScript changes → recompiles → restarts.
 */

import { spawn, ChildProcess, execSync } from 'child_process'
import { watch, existsSync, mkdirSync }  from 'fs'
import { join, resolve, relative }       from 'path'

let serverProcess: ChildProcess | null = null
let restartTimer:  ReturnType<typeof setTimeout> | null = null
let isCompiling:   boolean = false
let startCount:    number  = 0

const changedFiles: Set<string> = new Set()

const C = {
  reset:  '\x1b[0m',
  bold:   '\x1b[1m',
  dim:    '\x1b[2m',
  green:  '\x1b[32m',
  yellow: '\x1b[33m',
  red:    '\x1b[31m',
  cyan:   '\x1b[36m',
  gray:   '\x1b[90m'
}

function log(msg: string): void {
  const time = new Date().toLocaleTimeString('en-US', { hour12: false })
  console.log(`${C.gray}${time}${C.reset} ${C.cyan}[axil run]${C.reset} ${msg}`)
}

function logError(msg: string): void {
  const time = new Date().toLocaleTimeString('en-US', { hour12: false })
  console.log(`${C.gray}${time}${C.reset} ${C.red}[axil run]${C.reset} ${msg}`)
}

function logSuccess(msg: string): void {
  const time = new Date().toLocaleTimeString('en-US', { hour12: false })
  console.log(`${C.gray}${time}${C.reset} ${C.green}[axil run]${C.reset} ${msg}`)
}

function compile(cwd: string): boolean {
  const start = Date.now()

  try {
    execSync('npx tsc --incremental --noEmitOnError', {
      cwd,
      stdio: 'pipe'
    })

    const duration = Date.now() - start
    logSuccess(
      `Compiled ${C.bold}${changedFiles.size > 0 ? changedFiles.size + ' file(s)' : 'project'}${C.reset}${C.green} in ${C.bold}${duration}ms${C.reset}`
    )

    changedFiles.clear()
    return true
  } catch (error: any) {
    const duration = Date.now() - start
    const output   = error.stdout?.toString() || ''
    const stderr   = error.stderr?.toString() || ''

    logError(`Compilation failed in ${duration}ms`)

    const errorOutput = output || stderr
    if (errorOutput) {
      errorOutput.split('\n').forEach((line: string) => {
        if (line.includes('error TS')) {
          console.log(`  ${C.red}${line}${C.reset}`)
        } else if (line.trim()) {
          console.log(`  ${C.gray}${line}${C.reset}`)
        }
      })
    }

    logError('Server NOT restarted — fix the errors above')
    changedFiles.clear()
    return false
  }
}

function findEntryPoint(cwd: string): string | null {
  const candidates = [
    join(cwd, 'dist', 'main.js'),
    join(cwd, 'dist', 'server.js'),
    join(cwd, 'dist', 'index.js'),
    join(cwd, 'dist', 'src', 'main.js'),
    join(cwd, 'dist', 'src', 'server.js'),
    join(cwd, 'dist', 'src', 'index.js')
  ]

  for (const candidate of candidates) {
    if (existsSync(candidate)) return candidate
  }

  return null
}

async function startServer(cwd: string): Promise<void> {
  if (serverProcess) {
    serverProcess.kill('SIGTERM')

    await new Promise<void>(resolve => {
      const timer = setTimeout(() => {
        if (serverProcess) serverProcess.kill('SIGKILL')
        resolve()
      }, 2000)

      serverProcess!.once('exit', () => {
        clearTimeout(timer)
        resolve()
      })
    })

    serverProcess = null
  }

  const entry = findEntryPoint(cwd)

  if (!entry) {
    logError('Could not find entry point.')
    return
  }

  startCount++
  const relEntry = relative(cwd, entry)

  if (startCount === 1) {
    log(`Starting ${C.bold}${relEntry}${C.reset}`)
  } else {
    log(`Restarting ${C.bold}${relEntry}${C.reset} ${C.gray}(restart #${startCount - 1})${C.reset}`)
  }

  serverProcess = spawn('node', [entry], {
    cwd,
    stdio:  'inherit',
    env: {
      ...process.env,
      NODE_ENV:          process.env.NODE_ENV || 'development',
      FORCE_COLOR:       '1',
      NODE_NO_WARNINGS:  '1'
    }
  })

  serverProcess.on('error', (err) => {
    logError(`Failed to start server: ${err.message}`)
  })

  serverProcess.on('exit', (code, signal) => {
    if (signal === 'SIGTERM' || signal === 'SIGKILL') return
    if (code !== 0 && code !== null) logError(`Server exited with code ${code}`)
  })
}

function watchFiles(cwd: string): void {
  const srcDir = join(cwd, 'src')

  if (!existsSync(srcDir)) {
    logError(`src/ directory not found at ${srcDir}`)
    return
  }

  log(`Watching ${C.bold}src/${C.reset} for changes...`)

  watch(srcDir, { recursive: true }, (event, filename) => {
    if (!filename) return
    if (!filename.endsWith('.ts') && !filename.endsWith('.tsx')) return

    changedFiles.add(filename)

    if (restartTimer) clearTimeout(restartTimer)

    restartTimer = setTimeout(async () => {
      if (isCompiling) return
      isCompiling = true

      const files = Array.from(changedFiles)
      if (files.length === 1) {
        log(`Changed: ${C.yellow}${files[0]}${C.reset}`)
      } else {
        log(`Changed: ${C.yellow}${files.length} files${C.reset}`)
        files.forEach(f => console.log(`  ${C.gray}→ ${f}${C.reset}`))
      }

      const success = compile(cwd)

      if (success) {
        await startServer(cwd)
      }

      isCompiling = false
    }, 150)
  })
}

function validateProject(cwd: string): boolean {
  if (!existsSync(join(cwd, 'package.json'))) {
    logError('No package.json found.')
    return false
  }
  if (!existsSync(join(cwd, 'tsconfig.json'))) {
    logError('No tsconfig.json found.')
    return false
  }
  if (!existsSync(join(cwd, 'src'))) {
    logError('No src/ directory found.')
    return false
  }
  return true
}

export async function runDev(): Promise<void> {
  const cwd = process.cwd()

  console.log('')
  console.log(`${C.cyan}${C.bold} AxilJS${C.reset} ${C.gray}development server${C.reset}`)
  console.log(`${C.gray} ─────────────────────────────────────${C.reset}`)
  console.log('')

  if (!validateProject(cwd)) process.exit(1)

  const distDir = join(cwd, 'dist')
  if (!existsSync(distDir)) mkdirSync(distDir, { recursive: true })

  log('Initial compilation...')
  const success = compile(cwd)

  if (!success) {
    logError('Fix compilation errors then save a file to retry.')
    watchFiles(cwd)
    setupShutdown()
    return
  }

  await startServer(cwd)
  watchFiles(cwd)
  setupShutdown()
}

function setupShutdown(): void {
  process.on('SIGINT', async () => {
    console.log('')
    log('Shutting down...')

    if (serverProcess) {
      serverProcess.kill('SIGTERM')

      await new Promise<void>(resolve => {
        const timer = setTimeout(() => {
          serverProcess?.kill('SIGKILL')
          resolve()
        }, 3000)

        serverProcess!.once('exit', () => {
          clearTimeout(timer)
          resolve()
        })
      })
    }

    console.log('')
    process.exit(0)
  })
}

/**
 * axil test [pattern]
 * Discovers and runs test files.
 */

import { readdirSync, statSync } from 'fs'
import { join, resolve }         from 'path'

export async function runTests(pattern?: string): Promise<void> {
  const cwd     = process.cwd()
  const srcDir  = join(cwd, 'src')
  const testDir = join(cwd, 'tests')

  const testFiles: string[] = []

  if (statSync(srcDir, { throwIfNoEntry: false })?.isDirectory()) {
    scanDir(srcDir, testFiles)
  }

  if (statSync(testDir, { throwIfNoEntry: false })?.isDirectory()) {
    scanDir(testDir, testFiles)
  }

  if (testFiles.length === 0) {
    console.log('\x1b[33m[AxilJS Test]\x1b[0m No test files found.')
    return
  }

  const filtered = pattern
    ? testFiles.filter(f => f.includes(pattern))
    : testFiles

  if (filtered.length === 0) {
    console.log(`\x1b[33m[AxilJS Test]\x1b[0m No tests match: ${pattern}`)
    return
  }

  console.log(`\n\x1b[36m[AxilJS Test]\x1b[0m Found ${filtered.length} test file(s)\n`)

  try { require('ts-node/register') } catch {}

  for (const file of filtered) {
    try { require(resolve(file)) } catch (error) {
      console.error(`\x1b[31m[AxilJS Test]\x1b[0m Error loading ${file}:`, (error as Error).message)
    }
  }

  const { runTests: run } = require('@axiljs/testing')
  const passed = await run()

  process.exit(passed ? 0 : 1)
}

function scanDir(dir: string, results: string[]): void {
  const items = readdirSync(dir, { withFileTypes: true })
  for (const item of items) {
    const fullPath = join(dir, item.name)
    if (item.isDirectory() && item.name !== 'node_modules') {
      scanDir(fullPath, results)
    } else if (
      item.isFile() &&
      (item.name.endsWith('.test.ts') || item.name.endsWith('.spec.ts') ||
       item.name.endsWith('.test.js') || item.name.endsWith('.spec.js'))
    ) {
      results.push(fullPath)
    }
  }
}
